// Supabase Edge Function: يبعت إيميل لصاحبة المتجر عند كل أوردر جديد عبر Gmail (SMTP) // الإعدادات (حساب Gmail + كلمة مرور التطبيق + إيميل الاستلام) بتتحفظ من لوحة المالك داخل الموقع import { createClient } from 'https://esm.sh/@supabase/supabase-js@2'; import { SMTPClient } from 'https://deno.land/x/denomailer@1.6.0/mod.ts'; const cors = { 'Access-Control-Allow-Origin': '*', 'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type' }; const esc = (s: unknown) => String(s ?? '').replace(/[&<>"']/g, (c) => ({ '&': '&', '<': '<', '>': '>', '"': '"', "'": ''' }[c]!)); const PAY: Record = { cod: 'الدفع عند الاستلام', visa: 'فيزا عند الاستلام', instapay: 'تحويل InstaPay', vodafone: 'تحويل فودافون كاش' }; const reply = (t: string, status = 200) => new Response(t, { status, headers: cors }); Deno.serve(async (req) => { if (req.method === 'OPTIONS') return reply('ok'); try { const sb = createClient(Deno.env.get('SUPABASE_URL')!, Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!); const token = (req.headers.get('Authorization') || '').replace('Bearer ', ''); const { data: u } = await sb.auth.getUser(token); if (!u?.user) return reply('unauthorized', 401); const body = await req.json().catch(() => ({})); const { data: m } = await sb.from('mail_settings').select('*').eq('id', 1).maybeSingle(); if (!m?.gmail_user || !m?.gmail_app_password) return reply('لم يتم ضبط إعدادات الإيميل في لوحة المالك ← الإعدادات', 400); const to = m.notify_to || m.gmail_user; const send = async (subject: string, html: string) => { const c = new SMTPClient({ connection: { hostname: 'smtp.gmail.com', port: 465, tls: true, auth: { username: m.gmail_user, password: m.gmail_app_password } } }); await c.send({ from: m.gmail_user, to, subject, html }); await c.close(); }; // إيميل تجريبي (للأدمن فقط) if (body.test) { const { data: a } = await sb.from('admins').select('email').eq('email', u.user.email).maybeSingle(); if (!a) return reply('forbidden', 403); await send('✅ إيميل تجريبي من Velvet', '

إعدادات الإيميل شغالة، وهتوصلك إشعارات الأوردرات هنا.

'); return reply('sent'); } const { data: o } = await sb.from('orders').select('*, order_items(qty, unit_price, books(title))').eq('id', body.order_id).single(); // يرسل فقط لصاحب الأوردر نفسه ومرة واحدة if (!o || o.user_id !== u.user.id || o.notified) return reply('skip'); const items = (o.order_items as any[]).map((i) => `
  • ${esc(i.books?.title)} × ${i.qty} — ${i.unit_price} ج
    • `).join(''); const map = o.lat ? `https://www.google.com/maps?q=${o.lat},${o.lng}` : ''; await send(`🛍 أوردر جديد — ${o.total} ج`, `

    🛍 أوردر جديد على Velvet

      ${items}

    الإجمالي: ${o.total} ج
    طريقة الدفع: ${PAY[o.payment_method] ?? '-'}

    الاسم: ${esc(o.customer_name)}
    التليفون: ${esc(o.phone)}${o.phone2 ? ' / ' + esc(o.phone2) : ''}
    العنوان: ${esc(o.street)} - ${esc(o.place)}
    ${map ? `📍 الموقع على الخريطة` : ''}

    العميل هيكلمك على واتساب لتأكيد الأوردر.

    `); await sb.from('orders').update({ notified: true }).eq('id', o.id); return reply('sent'); } catch (e) { return reply(String(e), 500); } });
